Multivariable Models

Multiple Predictors and Omitted Variable Bias

In a single variable model, the coefficient is inflated. The observed variance is being overly attributed to the single variable. Once covariates are included in the model, then variance is more accurately attributed among them.

set.seed(2)
N <- 500 # sample size
skill <- rnorm(N) # skill variable
hours <- 2 + 0.9 * skill + rnorm(N, 0, 0.5) # hours variable, linearly related to skill
score <- 50 + 8 * skill + 1.5 * hours + rnorm(N, 0, 3) # score outcome variable, linearly related to skill and hours

round(c(naive    = coef(lm(score ~ hours))[2], # simple regression
        adjusted = coef(lm(score ~ hours + skill))[2], # adjusted regression
        truth    = 1.5), 3) # simulation true coefficient value for hours
##    naive.hours adjusted.hours          truth 
##          8.553          1.515          1.500

Once skill is included in the mode, the coefficient for hours is more reflective of the parameter.

plan  <- factor(rep(c("free", "pro", "ent"), each = 40),

set.seed(9)
plan  <- factor(rep(c("free", "pro", "ent"), each = 40),
                levels = c("free", "pro", "ent"))

# --- Part 1: your original model (factor only) -------------------------
spend0 <- c(rnorm(40, 50, 8), rnorm(40, 58, 8), rnorm(40, 66, 8))
fit0   <- lm(spend0 ~ plan)
round(coef(summary(fit0)), 3)
##             Estimate Std. Error t value Pr(>|t|)
## (Intercept)   50.166      1.193  42.054        0
## planpro        7.398      1.687   4.385        0
## planent       14.373      1.687   8.520        0
# The dummy columns R builds behind the scenes (one row per plan)
model.matrix(~ plan)[c(1, 41, 81), ]
##    (Intercept) planpro planent
## 1            1       0       0
## 41           1       1       0
## 81           1       0       1
# Intercept = mean of baseline (free); each dummy coefficient = shift from it
b0 <- coef(fit0)
rbind(from_coefs = c(free = b0[[1]],
                     pro  = b0[[1]] + b0[["planpro"]],
                     ent  = b0[[1]] + b0[["planent"]]),
      group_means = tapply(spend0, plan, mean))
##                 free      pro      ent
## from_coefs  50.16607 57.56377 64.53931
## group_means 50.16607 57.56377 64.53931
# --- Part 2: add a continuous predictor so the shifts become lines -----
usage <- runif(120, 0, 10)
shift <- c(free = 0, pro = 8, ent = 16)          # true intercept shifts
spend <- 40 + 2 * usage + shift[as.character(plan)] + rnorm(120, 0, 4)

fit <- lm(spend ~ usage + plan)
round(coef(summary(fit)), 3)
##             Estimate Std. Error t value Pr(>|t|)
## (Intercept)   40.383      0.893  45.216        0
## usage          1.952      0.123  15.861        0
## planpro        7.490      0.884   8.473        0
## planent       16.186      0.885  18.289        0
b    <- coef(fit)
ints <- c(free = b[["(Intercept)"]],
          pro  = b[["(Intercept)"]] + b[["planpro"]],
          ent  = b[["(Intercept)"]] + b[["planent"]])

cols <- c(free = "steelblue", pro = "darkorange", ent = "firebrick")

plot(usage, spend, col = cols[as.character(plan)], pch = 19,
     xlab = "usage", ylab = "spend",
     main = "Same slope, different intercepts")
for (p in names(ints)) abline(a = ints[[p]], b = b[["usage"]],
                              col = cols[[p]], lwd = 2)
legend("topleft", legend = names(cols), col = cols, pch = 19, lwd = 2,
       title = "plan")

# Vertical gap between lines at any usage value = the dummy coefficient
segments(x0 = 0, y0 = ints[["free"]], y1 = ints[["pro"]], lty = 2)
segments(x0 = 0, y0 = ints[["free"]], y1 = ints[["ent"]], lty = 3)
